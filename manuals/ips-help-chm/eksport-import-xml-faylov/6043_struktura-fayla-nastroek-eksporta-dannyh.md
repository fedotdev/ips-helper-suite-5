# Структура файла настроек экспорта данных
Главным элементом XML-документа является тег **< xmlexportsettings>**, который содержит всю информацию. Внутри основного элемента располагаются секции для описания типов атрибутов/параметров, объектов, связей, правил разворота состава соответствующие тегам **< attributes>, <object_types>, <relation_types >, <applsettings>.**
<!-- Секция с информацией о настройках экспорта -->
<xmlexportsettings ....>
<!-- Глобальный список экспортируемых атрибутов -->
<attributes>
< attribute .... />
.... 
</attributes>
<!-- Список экспортируемых типов объектов -->
<object_types>
<object_type .... />
....
</object_types>
<!-- Список экспортируемых типов связей -->
<relation_types>
<relation_type .... />
....
</relation_type>
<!-- Правила получения состава объектов при экспорте -->
<applsettings>
<applicability .... />
</applsettings>
